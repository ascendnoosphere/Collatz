\title{Classical Parent Joins and Terminal Words}
\author{}
\date{\today}

\begin{document}
\maketitle

\begin{abstract}
We give exact smaller-parent certificates for the classical shortcut map,
including certificates whose meeting point lies above the original seed.
Four elementary rules reduce capture to an explicitly described residual
set of natural density \(1/12\). Three further legal words reduce that set
to density \(11/216\). Two second-burst rules, uniform in the burst length, and two paired-tail
rules reduce the residual density to \(89419/2654208\). Eight explicit
direct-descent cylinders reduce it further to \(65659/2654208\). All
covered seeds retain uniform contraction of their original parent labels.
We also give an exact valuation budget for repetitions of any fixed word
and classify terminal powers within the initial prefix \(1^a01\).
Every reduction retains both orbit clocks and the strict parent inequality.
These results do not assert capture for the remaining residual seeds.
\end{abstract}

\section{Clocks and joins}

Throughout, the domain is the positive integers and
\[
T(n)=\begin{cases}n/2,&2\mid n,\\(3n+1)/2,&2\nmid n.\end{cases}
\]
A word records the parity of the input at each shortcut tick: \(1\) is the
odd branch and \(0\) is the even branch. A legal word of length \(h\) with
\(r\) odd letters expands to \(h+r\) division-first raw ticks.
The target shortcut cycle is \(1\to2\to1\).

% Join recovery must allow the parent to have reached one before the meeting.
\begin{lemma}[Join recovery]
If \(T^b(n)=T^c(p)\) and \(p\) reaches \(1\), then \(n\) reaches \(1\).
If, in addition, \(T^d(p)=T^e(q)\), then
\[
T^{b+d}(n)=T^{c+e}(q).
\]
\end{lemma}
\begin{proof}
Choose \(k\) with \(T^k(p)=1\). If \(c\le k\), another \(k-c\) steps from
the meeting point reach \(1\). If \(c>k\), the meeting point is \(1\) or
\(2\), since the target cycle is invariant; zero or one further steps
reach \(1\). This proves the first assertion without comparing the
meeting time to the first arrival time. Applying \(T^d\) to the first
join and \(T^c\) to the second proves the displayed composition identity.
\end{proof}

Thus any rule assigning \(1\le p<n\) and a legal join permits strong
induction on the original integer label. A join alone, without this
inequality, does not furnish that induction step.

\section{An explicit residual set}

For odd \(n\), write uniquely
\[
n=2^a u-1,\qquad a=v_2(n+1)\ge1,\qquad u>0\text{ odd}.
\]
Set
\[
\mathcal R=\{2^a u-1:a\ge2,\ u>0\text{ odd},\ 3\nmid u,
                         \ 3^a u\equiv3\pmod4\}.
\]

% The four rules are ordered so their failure determines exactly the residual class.
\begin{theorem}[Four parent rules]
Every \(n>1\) outside \(\mathcal R\) has a positive parent \(p<n\) with
\(5p\le4n\) and an explicit join. Apply these rules in order:
\begin{enumerate}[label=\arabic*.]
\item If \(n\) is even, take \(p=n/2\), so \(T(n)=p\).
\item If \(n\equiv1\pmod4\), take \(p=(3n+1)/4\), so \(T^2(n)=p\).
\item If \(n\equiv2\pmod3\), take \(p=(2n-1)/3\), so \(T(p)=n\).
\item If \(a\ge2\) and \(3^a u\equiv1\pmod4\), take
\(p=2^{a-1}u-1\), so \(T^{a+2}(n)=T^{a+1}(p)\).
\end{enumerate}
\end{theorem}
\begin{proof}
The first identity and inequality are immediate. In the second case,
\(n>1\) implies \(n\ge5\); the first output is even and
\(5(3n+1)\le16n\). In the third case, the quotient is a positive odd
integer, \(3p+1=2n\), and \(p<2n/3\).

For the fourth rule the exact initial odd run is
\[
T^j(n)=3^j2^{a-j}u-1\quad(0\le j\le a).
\]
For \(j<a\) the input is odd, which proves the formula inductively;
the endpoint at \(j=a\) is even. The assumed congruence gives two even
steps from that endpoint, so
\[
T^{a+2}(n)=\frac{3^a u-1}{4}.
\]
The parent's initial run has length \(a-1\). Its endpoint is
\(3^{a-1}u-1\), congruent to \(2\) modulo \(4\). One even step and one
odd step therefore give
\[
T^{a+1}(p)=\frac{3(3^{a-1}u-1)+2}{4}
          =\frac{3^a u-1}{4}.
\]
The parent is positive because \(a\ge2\), and \(2p=n-1\).
If rules one and two fail, \(a\ge2\). If rule three also fails, then
\(3\nmid u\), since \(3\mid u\) would imply \(n\equiv2\pmod3\).
The odd integer \(3^a u\) is \(1\) or \(3\) modulo \(4\); failure of the
fourth rule leaves precisely \(\mathcal R\).
\end{proof}

\section{Three further residual certificates}

% These are coefficient identities on entire cylinders, not representative tests.
\begin{lemma}[Three cylinder identities]
For every integer \(t\ge0\),
\[
\begin{aligned}
T^3(3+8t)&=4+9t,\\
T^5(11+32t)&=10+27t,\\
T^6(7+64t)&=10+81t.
\end{aligned}
\]
The parent words in the first and third identities are \(110\) and
\(111010\); the source word in the second is \(11010\).
\end{lemma}
\begin{proof}
The first two state lists are
\[
3+8t\to5+12t\to8+18t\to4+9t
\]
and
\[
\begin{aligned}
11+32t&\to17+48t\to26+72t\\
      &\to13+36t\to20+54t\to10+27t.
\end{aligned}
\]
Every parity guard follows from the displayed constant term and even
coefficient at that input. Exact substitution in the two branches proves
every arrow. In the first list the parent satisfies
\(9(3+8t)<8(4+9t)\). In the second,
\(10+27t<11+32t\), and for \(t\ge1\) also
\(9(10+27t)\le8(11+32t)\).
The third state list is
\[
\begin{aligned}
7+64t&\to11+96t\to17+144t\to26+216t\\
     &\to13+108t\to20+162t\to10+81t.
\end{aligned}
\]
The constant parities verify \(111010\), and every coefficient is even
at its input. Substitution proves each step. Its positive parent obeys
\(9(7+64t)<8(10+81t)\), so is strictly smaller than the source.
\end{proof}

Define the smaller residual set
\[
\mathcal R_*=
\left\{n\in\mathcal R:
\begin{array}{l}
n\not\equiv4\pmod9,\quad n\not\equiv11\pmod{32},\\
n\not\equiv10\pmod{81}
\end{array}\right\}.
\]

\begin{theorem}[Residual recovery]
Every \(n>1\) outside \(\mathcal R_*\) admits a positive smaller-parent
join with \(9p\le8n\). Repeated parent reduction terminates at \(1\) or
in \(\mathcal R_*\), after at most \(\log n/\log(9/8)\) reductions.
Capture of every member of \(\mathcal R_*\) implies capture of every
positive integer. Moreover, \(d(\mathcal R_*)=11/216\).
\end{theorem}
\begin{proof}
Outside \(\mathcal R\), use the four parent rules; their factor \(4/5\)
is at most \(8/9\). Inside \(\mathcal R\), if \(n\equiv4\pmod9\), write
\(n=4+9t\) and use parent \(3+8t\) with source time zero. Otherwise, if
\(n\equiv11\pmod{32}\), write \(n=11+32t\) and use the five-tick
endpoint as parent. Here \(t=0\) would give \(u=3\), contrary to
\(3\nmid u\), so the uniform eight-ninths bound applies.
For the final uncovered class \(n\equiv10\pmod{81}\), write
\(n=10+81t\) and use parent \(7+64t\) with source time zero and
parent time six. This certificate has the same uniform bound.
All parents stay positive. The natural-number label decreases strictly,
so iteration terminates. After \(k\) reductions its positive value is at
most \((8/9)^k n\), proving the logarithmic bound. Strong induction,
using join recovery and the assumed capture on \(\mathcal R_*\), proves
the capture implication. This bounds parent depth, not forward time.

For each fixed \(a\ge2\), the residual constraints select two classes
of \(u\) modulo \(12\), and hence density \(1/(6\cdot2^a)\) in the
original \(n\)-labels. The portion with \(a\ge A\) lies in
\(n\equiv-1\pmod{2^A}\) and has upper density at most \(2^{-A}\).
Finite sums followed by this tail bound give \(d(\mathcal R)=1/12\).
Adding \(n\equiv4\pmod9\) selects one class of \(u\) modulo \(36\)
for each \(a\); the condition \(3\nmid u\) is then automatic. Its
intersection with \(\mathcal R\) has density
\[
\sum_{a\ge2}\frac1{36\cdot2^a}=\frac1{72}.
\]
The class \(n=11+32t\) has \(a=2\) and \(u=3+8t\), which always has
the required congruence modulo \(4\). Excluding \(3\mid u\) removes
exactly \(t\equiv0\pmod3\); the intersection has density \(1/48\).
Finally \(11+32t\equiv4\pmod9\) means \(t\equiv4\pmod9\), giving
intersection density \(1/288\). These two exclusions leave density
\(1/12-1/72-1/48+1/288=5/96\).
For the third class \(n\equiv10\pmod{81}\), the congruence on
\(u\) modulo \(81\) is \(2^a u\equiv11\); it automatically
excludes \(3\mid u\). Together with the required class modulo four,
this selects one class modulo \(324\) for each fixed \(a\). Its
intersection with \(\mathcal R\) therefore has density
\[
\sum_{a\ge2}\frac1{324\cdot2^a}=\frac1{648}.
\]
It is disjoint from \(n\equiv4\pmod9\), since \(10\equiv1\pmod9\).
Its intersection with \(n\equiv11\pmod{32}\) has density
\(1/(81\cdot32)=1/2592\): the two coprime congruences select one
class, which has \(a=2\), the required odd-burst congruence, and
\(3\nmid u\). The valuation tail bound above still justifies all
infinite sums. Thus the final density is
\[
\frac5{96}-\frac1{648}+\frac1{2592}=\frac{11}{216}.
\]
\end{proof}

A genuinely residual example is the entire cylinder
\[
n=31+1152t,\qquad p=27+1024t\quad(t\ge0).
\]
Here \(a=5\), \(u=1+36t\), and \(T^3(p)=n\) with \(p<n\).
The source's first five ticks are odd and strictly increasing.
This is a parent certificate before any source descent. It also shows
why an increase of the source burst length does not itself prevent a
smaller-parent join. No assertion is made that every member of
\(\mathcal R_*\) has such a certificate.

\section{Uniform second-burst parents}

These rules permit unbounded source and parent clocks while keeping their
difference fixed. They are not finite lists of tested residue classes.
For \(a\in\{2,3\}\), let \(\mathcal S_a\) consist of the positive sources
\(n=2^a u-1\) for which
\[
\begin{gathered}
u>0\text{ odd},\qquad b\ge1,\qquad v>0\text{ odd},\\
3^a u+1=2^{b+1}v,\qquad 3^b v\equiv1\pmod4.
\end{gathered}
\]
The odd factor \(v\) determines \(b\) uniquely. The entrance equation
also gives \(3^a u\equiv3\pmod4\), as required by the original residual
apart from its restriction modulo three.

% The shorter parent run improves contraction and admits a third initial odd letter.
\begin{theorem}[Two- and three-burst joins]
For \(a\in\{2,3\}\) and every \(n\in\mathcal S_a\), the positive integer
\[
p=2^{b-1}v-1=\frac{3^a u-3}{4}
\]
satisfies
\[
T^{a+b+3}(n)=T^{b+1}(p),\qquad p<n,\qquad 7p\le6n.
\]
The source and parent words are respectively \(1^a01^b00\) and
\(1^{b-1}01\). Their common endpoint is \((3^b v-1)/4\).
More precisely,
\[
\begin{array}{c|c|c}
a&\text{exact original-source relation}&\text{contraction}\\
\hline
2&16p+3=9n&16p\le9n\\
3&32p=27n+3&7p\le6n.
\end{array}
\]
\end{theorem}
\begin{proof}
The odd-run formula gives
\[
T^j(n)=3^j2^{a-j}u-1\quad(0\le j\le a).
\]
Since \(u\) is odd, the first \(a\) input guards are odd, and their
endpoint \(3^a u-1\) is even. Its next halving gives
\[
T^{a+1}(n)=\frac{3^a u-1}{2}=2^b v-1.
\]
Because \(v\) is odd, this state has precisely \(b\) initial odd guards.
The second odd run therefore ends at
\[
T^{a+b+1}(n)=3^b v-1.
\]
The assumed congruence makes this endpoint divisible by four.
Two legal halvings give \(T^{a+b+3}(n)=(3^b v-1)/4\).

For the parent, \(p+1=2^{b-1}v\), so its initial odd run has length
\(b-1\) and ends at \(3^{b-1}v-1\).
The source guard implies \(3^{b-1}v\equiv3\pmod4\).
Thus this parent endpoint is two modulo four; a halving is followed by
an odd shortcut. Their result is
\[
T^{b+1}(p)=\frac{3(3^{b-1}v-1)+2}{4}
         =\frac{3^b v-1}{4}.
\]
When \(b=1\), the initial parent run is empty, and its initial state
itself is two modulo four. The same guarded two-step calculation applies.

The entrance equation gives \(4(p+1)=3^a u+1\), hence the displayed
formula for \(p\). Its numerator is at least six for \(a=2\) and
at least twenty-four for \(a=3\), so the integral parent is positive.
For \(a=2\), substitution of \(n+1=4u\) gives \(16p+3=9n\);
this proves \(p<n\), \(16p\le9n\), and \(7p\le6n\).
For \(a=3\), substitution of \(n+1=8u\) gives \(32p=27n+3\).
Here \(n\ge7\), so \(27n+3<32n\) and
\(7(27n+3)\le192n\), proving the remaining bounds.
Both comparisons are against the original source, not the larger
meeting point.
\end{proof}

Two entire residual multiple-of-three cylinders illustrate the joins:
\[
\begin{aligned}
T^9(411+1536t)&=T^5(231+864t)=587+2187t,\\
T^{10}(615+3072t)&=T^5(519+2592t)=1316+6561t
\qquad(t\ge0).
\end{aligned}
\]
For the first, take \(a=2\), \(b=4\), \(u=103+384t\),
and \(v=29+108t\). For the second, take \(a=3\), \(b=4\),
\(u=77+384t\), and \(v=65+324t\).
These substitutions verify the entrance equations and the terminal guards.
Their respective sources are \(27\) and \(7\) modulo \(32\);
they are multiples of three, while \(u\equiv1\) and \(2\) modulo three
respectively. Thus neither cylinder is removed by any earlier exclusion,
and both source cylinders lie in \(\mathcal R_*\).
At \(t=0\), every positive source tick through its displayed join
lies above its original source. For example, the second source runs
\[
\begin{aligned}
615&\to923\to1385\to2078\to1039\to1559\\
   &\to2339\to3509\to5264\to2632\to1316,
\end{aligned}
\]
while \(519\to779\to1169\to1754\to877\to1316\).
These are smaller-parent certificates before any source descent.

% Exact dyadic tails justify the count over all second-burst lengths.
\begin{theorem}[Refined residual density and recovery]
Set
\[
\mathcal R_{**}=\mathcal R_*\setminus(\mathcal S_2\cup\mathcal S_3).
\]
Then
\[
d(\mathcal R_{**})=\frac{11}{324},\qquad
d(\mathcal R_{**}\cap3\Z)=\frac1{48}.
\]
Every \(n>1\) outside this set has a positive smaller-parent join with
\(9p\le8n\). Capture of every member of \(\mathcal R_{**}\) implies
capture of every positive integer, with the same logarithmic parent-depth
bound as before. For each \(a=2,3\),
\[
d(\mathcal R_*\cap\mathcal S_a)=\frac{11}{1296}.
\]
Exactly half of each previously residual part with \(v_2(n+1)=2\) or
\(v_2(n+1)=3\), measured by natural density, is removed.
\end{theorem}
\begin{proof}
Fix \(a\in\{2,3\}\) and \(b\ge1\).
The terminal congruence specifies one odd class of \(v\) modulo four.
The entrance equation therefore specifies one class of \(u\) modulo
\(2^{b+3}\), because \(3^a\) is invertible modulo that power of two.
It selects one class of \(n\) modulo \(2^{a+b+3}\), of density
\(2^{-(a+b+3)}\). Different \(b\) give disjoint classes.
For \(b\ge B\ge1\), all these sources satisfy
\(3^a u+1\equiv0\pmod{2^{B+1}}\).
They lie in one class modulo \(2^{a+B+1}\), of upper density at most
\(2^{-(a+B+1)}\). This tends to zero and justifies every infinite sum below.

For \(a=2\), the class with \(b=1\) has \(u\equiv3\pmod8\), hence
\(n\equiv11\pmod{32}\), and is already excluded from \(\mathcal R_*\).
The remaining classes, with \(b\ge2\), have \(u\equiv7\pmod8\)
and \(n\equiv27\pmod{32}\). Their combined density is
\[
\sum_{b\ge2}2^{-(b+5)}=\frac1{64}.
\]
For \(a=3\), the entrance equation gives \(u\equiv1\pmod4\),
hence \(n\equiv7\pmod{32}\). No such class meets the exclusion
\(n\equiv11\pmod{32}\), and their combined density is
\[
\sum_{b\ge1}2^{-(b+6)}=\frac1{64}.
\]

For either remaining family, membership in \(\mathcal R\) adds only
\(3\nmid u\). The Chinese remainder theorem retains two out of three
classes at each fixed \(b\). The exclusions \(n\equiv4\pmod9\)
and \(n\equiv10\pmod{81}\) respectively occupy fractions \(1/9\)
and \(1/81\) of each dyadic class. They already imply \(3\nmid u\)
and are disjoint, since \(10\pmod{81}\) is one, not four, modulo nine.
Consequently, for each \(a=2,3\),
\[
d(\mathcal R_*\cap\mathcal S_a)
 =\left(\frac23-\frac19-\frac1{81}\right)\frac1{64}
 =\frac{11}{1296}.
\]
The two families have different exact valuations of \(n+1\), so are
disjoint. Subtracting both removals from \(11/216\) gives \(11/324\).
The whole \(a=2\) part of \(\mathcal R_*\) is the class
\(n\equiv27\pmod{32}\) with the same three odd-modulus restrictions.
The whole \(a=3\) part is \(n\equiv7\pmod{32}\) with those restrictions.
Each therefore has density
\[
\left(\frac23-\frac19-\frac1{81}\right)\frac1{32}
=\frac{11}{648},
\]
twice its removed density.

For multiples of three, each remaining burst family has density
\(1/(3\cdot64)=1/192\). The condition \(3\nmid u\) is automatic there,
and neither odd-modulus exclusion contains a multiple of three.
The original multiple-of-three part of \(\mathcal R\) has density
\(\sum_{a\ge2}1/(12\cdot2^a)=1/24\).
The class \(n\equiv11\pmod{32}\) removes density \(1/96\):
its sources \(11+32t\) are divisible by three exactly when
\(t\equiv2\pmod3\). Hence
\[
d(\mathcal R_*\cap3\Z)=\frac1{24}-\frac1{96}=\frac1{32},\qquad
d(\mathcal R_{**}\cap3\Z)=\frac1{32}-\frac2{192}=\frac1{48}.
\]

Outside \(\mathcal R_*\), use the previous parent certificates.
Inside either newly removed family, use the burst theorem.
Its factor \(6/7\) is below \(8/9\).
The parent remains a positive integer smaller than the original seed.
Strong induction and join recovery prove the capture implication.
Iteration terminates because the integer label decreases at every choice.
After \(j\) choices the positive label is at most \((8/9)^j n\);
hence \(j\le\log n/\log(9/8)\).
This is a parent-depth bound, not a forward stopping-time bound.
No capture on the remaining residual is asserted.
\end{proof}

\section{Paired eight-letter tails}

The next construction changes the tail rather than reusing the two-letter
ending. It keeps the common denominator but compensates for a shorter
initial odd run on the parent side with more odd letters in its tail.
For a rational word map \(F_w(x)=(3^{r(w)}x+B_w)/2^h\), define its
centered offset to be \(C_w=B_w-3^{r(w)}\). Then
\[
F_w(3^b v-1)=\frac{3^{b+r(w)}v+C_w}{2^h}.
\]
Thus equal lengths and equal centered offsets give a potential join
when the parent odd count exceeds the source odd count by the number
of letters removed from its initial run. Legality still has to be proved.

% The coefficient table records both the affine maps and their exact legal cylinders.
\begin{lemma}[Three compatible tails]
These length-eight words have centered offset \(125\).
Their legal input classes are as follows:
\[
\begin{array}{c|c|c}
w&F_w(x)&x\pmod{256}\\
\hline
00000001&(3x+128)/256&128\\
01000001&(9x+134)/256&42\\
01011100&(81x+206)/256&146.
\end{array}
\]
\end{lemma}
\begin{proof}
Starting with \((P,Q,B)=(1,1,0)\), each even letter replaces this
triple by \((P,2Q,B)\), and each odd letter by
\((3P,2Q,3B+Q)\). Eight substitutions give the three displayed maps.
Subtracting \(3\), \(9\), and \(81\) from their respective offsets
gives \(125\). Each numerator is divisible by \(256\) exactly on
its displayed input class, since its multiplier is odd.
The integer cylinder criterion proved in Lemma~\ref{lem:IntegerCylinder}
below makes this final integrality equivalent to legality of every prefix.
Alternatively, the inputs at the class representatives run as follows:
\[
\begin{aligned}
128&\to64\to32\to16\to8\to4\to2\to1\to2,\\
42&\to21\to32\to16\to8\to4\to2\to1\to2,\\
146&\to73\to110\to55\to83\to125\to188\to94\to47.
\end{aligned}
\]
Adding \(256t\) at the input preserves all eight guards: at the
\(j\)-th prefix the added amount is \(3^s2^{8-j}t\), which is even
for \(j<8\). This also proves the class assertions directly.
\end{proof}

For \(a=4,5,6\), let \(\mathcal H_a\) contain the positive sources
\(n=2^a u-1\) with
\[
\begin{gathered}
u>0\text{ odd},\quad b\ge3,\quad v>0\text{ odd},\\
3^a u+1=2^{b+1}v,\qquad 3^b v\equiv129\pmod{256}.
\end{gathered}
\]
For \(a=2,3,4\), let \(\mathcal J_a\) be defined instead by
\[
\begin{gathered}
u>0\text{ odd},\quad b\ge2,\quad v>0\text{ odd},\\
3^a u+1=2^{b+1}v,\qquad 3^b v\equiv43\pmod{256}.
\end{gathered}
\]
The first ending has terminal residue one modulo four; the second has
terminal residue three modulo four, so it reaches sources not covered
by the earlier second-burst ending.

% Every comparison uses the original source label, with no bound on the second run.
\begin{theorem}[Paired-tail parents]
For \(n\in\mathcal H_a\), set
\[
p=2^{b-3}v-1=\frac{3^a u-15}{16}.
\]
Then the words \(1^a01^b00000001\) and \(1^{b-3}01011100\) give
\[
T^{a+b+9}(n)=T^{b+5}(p)=\frac{3^{b+1}v+125}{256},\qquad
0<p<n,\qquad1024p\le729n.
\]
For \(n\in\mathcal J_a\), set
\[
p=2^{b-2}v-1=\frac{3^a u-7}{8}.
\]
The words \(1^a01^b01000001\) and \(1^{b-2}01011100\) give
\[
T^{a+b+9}(n)=T^{b+6}(p)=\frac{3^{b+2}v+125}{256},\qquad
0<p<n,\qquad128p\le81n.
\]
Neither family has an upper bound on \(b\). The parent run is empty
when \(b=3\) in the first family or \(b=2\) in the second.
\end{theorem}
\begin{proof}
The entrance equation and odd-run formula give
\[
T^{a+1}(n)=2^b v-1,\qquad T^{a+b+1}(n)=3^b v-1.
\]
The first \(a\) and next \(b\) odd letters are legal because \(u\)
and \(v\) are odd; the intervening endpoint is even. In the first
family, the source tail input is \(128\) modulo \(256\).
The parent reaches \(y=3^{b-3}v-1\) after \(b-3\) odd letters.
Its shifted input obeys
\(27(y+1)\equiv129\pmod{256}\), hence
\(y\equiv146\pmod{256}\). The lemma proves legality of both tails.
Their endpoint numerators are respectively
\[
3(3^b v-1)+128=3^{b+1}v+125,
\qquad81(3^{b-3}v-1)+206=3^{b+1}v+125.
\]
This proves the first join, including both clocks.

% A vanishing dyadic tail controls the infinite count before CRT restrictions are imposed.
\begin{theorem}[Residual after paired-tail removal]
Put
\[
\mathcal H=\bigcup_{a=4}^6\mathcal H_a,\qquad
\mathcal J=\bigcup_{a=2}^4\mathcal J_a,\qquad
\mathcal R_{\dagger}=\mathcal R_{**}\setminus(\mathcal H\cup\mathcal J).

Then
\[
d(\mathcal R_{\dagger})=\frac{89419}{2654208},\qquad
 d(\mathcal R_{\dagger}\cap3\Z)=\frac{8129}{393216}.
\]
Every \(n>1\) outside \(\mathcal R_{\dagger}\) has a positive
smaller-parent join with \(9p\le8n\). Parent reduction terminates at
one or in \(\mathcal R_{\dagger}\) in at most
\(\log n/\log(9/8)\) reductions. Capture on this residual implies
universal positive capture; capture on the residual is not proved.
\end{theorem}
\begin{proof}
Fix an allowed \(a,b\). The guard selects one odd class of \(v\)
modulo \(256\), since \(3^b\) is a unit modulo \(256\).
The entrance equation then selects one class of \(u\) modulo
\(2^{b+9}\), and hence one class of \(n\) modulo \(2^{a+b+9}\).
Every member has the exact initial rank \(a\) and second rank \(b\).
Different ranks therefore give disjoint classes.
For \(b\ge B\), the entrance puts all sources in one class modulo
\(2^{a+B+1}\). Its upper density is at most \(2^{-(a+B+1)}\),
which tends to zero. Finite sums with this tail bound give
\[
\begin{aligned}

\end{aligned}
\]
The union of \(\mathcal H\) and \(\mathcal J\) is disjoint: the only
shared initial rank is four, and its terminal guards are different.
Neither family meets \(\mathcal S_2\cup\mathcal S_3\).
For \(\mathcal H\), the initial ranks differ; for \(\mathcal J\),
the terminal residue is three rather than one modulo four.

\[

 d(\mathcal R_{**}\cap\mathcal J)=\frac{77}{331776}.

\[

\]
Subtracting it from \(1/48\) gives \(8129/393216\).

parent strictly below its original seed, with \(9p\le8n\).

prove the capture implication. The natural-number label decreases
at each parent choice, and after \(j\) choices it is at most
\((8/9)^j n\). Positivity gives the stated depth bound.
This remains a conditional reduction, not capture of the residual.
\end{proof}

% This lemma converts one checked base trajectory into a uniform symbolic cylinder.
\begin{lemma}[Dyadic cylinder lifting]\label{lem:DyadicLifting}
Suppose the first \(h\) shortcut inputs on the trajectory from \(c\) have
exactly \(r\) odd members. Then, for every \(t\ge0\), those \(h\) parity
decisions are unchanged after replacing \(c\) by \(c+2^ht\), and
\[

\]
\end{lemma}
\begin{proof}

\(j\) base steps. Induction on \(j\) gives
\[
T^j(c+2^ht)=T^j(c)+3^{r_j}2^{h-j}t
\qquad(0\le j\le h).
\]
For \(j<h\), the added term is even, so the next parity agrees with that of
the base state. An even step divides the added term by two, while an odd
step multiplies it by three and then divides it by two. These are exactly
the two induction updates. At \(j=h\), the formula is the claimed identity.
\end{proof}

It leaves forty-four classes modulo \(81\), proving the third density.

\end{proof}
\section{Exact repeated-word budget}

For a nonempty binary word \(w\) of length \(h\), compose the rational
branches to obtain
\[
F_w(n)=\frac{Pn+B}{Q},\qquad P=3^{r(w)},\quad Q=2^h.
\]
An appended odd letter replaces \((P,Q,B)\) by
\((3P,2Q,3B+Q)\); an even letter replaces it by \((P,2Q,B)\).

% The legality criterion uses the odd suffix multiplier to recover every prefix.
\begin{lemma}[Integer cylinder criterion]\label{lem:IntegerCylinder}
The word \(w\) is legal at an integer \(n\) if and only if \(Q\mid Pn+B\).
\end{lemma}
\begin{proof}
Legality implies integrality of the endpoint. Conversely, fix a prefix
of length \(j\) with numerator \(N_j\) and denominator \(2^j\).
The full numerator is \(3^sN_j+2^jC\), where \(s\) and \(C\) come from
the suffix. Divisibility by \(2^h\) therefore implies \(2^j\mid N_j\),
since \(3^s\) is odd. Thus every rational prefix state is integral.
For an even letter, integrality of its successor forces its input even.
For an odd letter, integrality of \((3x+1)/2\) forces its input odd.
Every letter is therefore the actual shortcut branch.
\end{proof}

\begin{theorem}[Repetition budget]
Put \(D_w(n)=(P-Q)n+B\). For every \(k\ge0\),
\[
w^k\text{ is legal at }n\quad\Longleftrightarrow\quad Q^k\mid D_w(n).
\]
If \(D_w(n)\ne0\), the maximal number of complete initial copies is
\[
\left\lfloor\frac{v_2(|D_w(n)|)}h\right\rfloor.
\]
\end{theorem}
\begin{proof}
The numerator of the \(k\)-fold rational composition is
\[
N_k=P^kn+B\sum_{j=0}^{k-1}P^{k-1-j}Q^j.
\]
Multiplication and telescoping give
\[
(P-Q)N_k=P^kD_w(n)-BQ^k.
\]
Both \(P\) and \(P-Q\) are odd, so each is a unit modulo \(Q^k\).
Consequently \(Q^k\mid N_k\) if and only if \(Q^k\mid D_w(n)\).
The cylinder criterion supplies legality. The valuation statement
follows since \(Q^k=2^{hk}\). For \(k=0\) the legality and divisibility
conditions are both automatic.

The Lean proof uses the same budget through a one-copy transport identity.
For a legal copy with endpoint \(m=F_w(n)\), the affine equation gives
\[
D_w(n)=Q(m-n),\qquad D_w(m)=P(m-n).
\]
Since \(P\) is odd, it is coprime to \(Q^k\). Cancelling these factors
shows that \(Q^{k+1}\mid D_w(n)\) is equivalent to
\(Q^k\mid D_w(m)\). The cylinder criterion supplies the first legal copy,
and induction on \(k\) proves the repetition criterion. This also covers
the empty word when \(P=Q=1\) and \(B=0\); the valuation formula requires
\(h>0\) as stated.
\end{proof}

For \(w=110\), one has \((P,Q,B)=(9,8,5)\), hence the budget is
\(\lfloor v_2(n+5)/3\rfloor\). For a positive seed and \(P>Q\),
\(D_w(n)>0\), because \(B\ge0\); such a word cannot repeat forever.
The theorem concerns repetitions of one fixed word, not switching
between different words.

\section{A bounded-depth obstruction}

% This obstructs finite dyadic covers using only the two certificate types below.
For \(t\ge0\), put \(n(t)=27+1024t\). Its first two valuation blocks satisfy
\[
n(t)=2^2(7+256t)-1,
\qquad
9(7+256t)+1=2^6(1+36t).
\]
Thus \(a=2\), \(b=5\), and \(v=1+36t\). Substitution in all the
guards defining \(\mathcal R_{\mathrm D}\) gives the exact test
\begin{equation}\label{eq:TwentySevenResidual}
n(t)\in\mathcal R_{\mathrm D}
\quad\Longleftrightarrow\quad
\begin{cases}
t\not\equiv2\pmod3,\\
t\not\equiv7\pmod9,\\
t\not\equiv76\pmod{81},\\
t\not\equiv42\pmod{64}.
\end{cases}
\end{equation}
Indeed, the first three conditions are respectively \(3\nmid u\),
\(n\not\equiv4\pmod9\), and \(n\not\equiv10\pmod{81}\). The
second-burst guard fails because \(3^5v\equiv3\pmod4\), while the only
applicable paired-tail guard is
\[
3^5v\equiv43\pmod{256}
\quad\Longleftrightarrow\quad t\equiv42\pmod{64}.
\]
The class \(27\pmod{1024}\) is in none of the eight direct cylinders.

For \(L\ge1\), let \(\tau_L\) be the unique residue modulo \(2^L\)
such that
\begin{equation}\label{eq:TauDefinition}
183+6561\tau_L\equiv0\pmod{2^L}.
\end{equation}
It exists uniquely because \(6561\) is odd. For \(L\ge6\), reduction
modulo \(64\) gives \(\tau_L\equiv41\pmod{64}\). Define
\[
\mathcal E_L=
\left\{27+1024t:
\begin{array}{l}
t\equiv\tau_L\pmod{2^L},\\
t\not\equiv2\pmod3,\\
t\not\equiv7\pmod9,\\
t\not\equiv76\pmod{81}
\end{array}
\right\}.
\]
For \(1\le d\le5\), put
\[
\begin{aligned}
p_d(t)&=2^{5-d}(1+36t)-1,\\
X(t)&=242+8748t,
&Y_d(t)&=3^{5-d}(1+36t)-1.
\end{aligned}
\]
We allow the empty word, which has length zero, is legal at every integer,
has \(F_{\varnothing}\) equal to the identity, and satisfies
\(r(\varnothing)=0\).

\begin{theorem}[Bounded-depth obstruction]\label{thm:BoundedDepthObstruction}
For every \(H\ge1\), choosing \(L\ge\max\{6,H\}\) gives an infinite
set \(\mathcal E_L\subset\mathcal R_{\mathrm E}\) of density
\[
d(\mathcal E_L)=\frac{44}{81\cdot2^{L+10}}
\]
with the following properties.
\begin{enumerate}
\item No \(n\in\mathcal E_L\) has \(T^j(n)<n\) for
      \(1\le j\le H\).
\item None of the five smaller parents \(p_d(t)\) admits a uniform
affine-tail collision with the source using fixed words of lengths at most
\(H\) that are legal at their respective source and parent inputs on any
nonempty intersection of \(\mathcal E_L\) with a parameter congruence
\(t\equiv c\pmod{2^M}\).
\end{enumerate}
More precisely, put
\[
C=\{t\ge0:n(t)\in\mathcal R_{\mathrm E}\}.
\]
There is no finite cover of \(C\) by sets
\[
C_i=C\cap\{t:t\equiv c_i\pmod{2^{M_i}}\},
\qquad M_i\ge0,
\]
such that every \(C_i\) has one fixed certificate of either form:
\begin{enumerate}[label=(\alph*)]
\item a time \(j_i\ge1\) for which \(T^{j_i}(n(t))<n(t)\) throughout
      \(C_i\); or
\item a number \(d_i\in\{1,\ldots,5\}\), two possibly empty words
      \(w_{s,i},w_{p,i}\) of respective lengths \(h_{s,i},h_{p,i}\),
      such that for every \(t\in C_i\), the words are legal at
      \(X(t),Y_{d_i}(t)\), respectively, and
\[
F_{w_{s,i}}(X(t))=F_{w_{p,i}}(Y_{d_i}(t)).
\]
\end{enumerate}
\end{theorem}
\begin{proof}
The first ten shortcut iterates are
\[
\begin{aligned}
T^0(n(t))&=27+1024t,  &T^1(n(t))&=41+1536t,\\
T^2(n(t))&=62+2304t, &T^3(n(t))&=31+1152t,\\
T^4(n(t))&=47+1728t, &T^5(n(t))&=71+2592t,\\
T^6(n(t))&=107+3888t,&T^7(n(t))&=161+5832t,\\
T^8(n(t))&=242+8748t,&T^9(n(t))&=121+4374t,\\
T^{10}(n(t))&=182+6561t.&&
\end{aligned}
\]
Every entry after the first exceeds \(n(t)\). If
\(t\equiv\tau_L\pmod{2^L}\), write
\(183+6561t=2^Lq\). Then, for \(0\le j\le L\),
\[
T^{10+j}(n(t))=3^j2^{L-j}q-1.
\]
For \(0\le j<L\), these states are odd, and the \(L\) transitions from
\(j\) to \(j+1\) are increasing odd-branch steps. This proves the first
assertion.

The congruence \(\tau_L\equiv41\pmod{64}\) avoids the last exclusion in
\eqref{eq:TwentySevenResidual}. The other three exclusions leave
\(81-27-9-1=44\) classes modulo \(81\). They are disjoint: the first
has residue \(2\pmod3\), whereas the latter two have residue
\(1\pmod3\); those latter classes reduce respectively to \(7\) and
\(4\pmod9\). Since \(2^L\) is invertible modulo \(81\), the Chinese
remainder theorem proves infinitude,
inclusion in \(\mathcal R_{\mathrm D}\), and the stated density.
The source residue is \(27\pmod{1024}\), which is absent from every
residue listed for \(\mathcal A\). Membership in the \(2075\)-cylinder
would force \(t\equiv2\pmod8\), whereas
\(t\equiv\tau_L\equiv41\pmod{64}\). Membership in the deep
\(27\)-cylinder would force \(2^{49}\mid t\), whereas \(\tau_L\) is
odd. Finally, the \(111\)-join cylinder has source residue
\(111\pmod{1024}\). Thus \(\mathcal E_L\) misses all twenty-four
further source families, and hence
\(\mathcal E_L\subset\mathcal R_{\mathrm E}\).

It remains to audit the current parent mechanism. After the source's two valuation blocks, its tail starts at
\(X(t)=3^5v-1\).
Preserving \(v\) and shortening this second odd run by \(d\) produces
exactly the five parent forms in the statement. Every parameter in
\(\mathcal E_L\) satisfies \(t\ge41\), and hence
\[
0<p_5\le p_d\le p_1=15+576t<27+1024t=n(t).
\]
The choice \(d=0\) instead gives \(31+1152t>n(t)\), while \(d>5\)
would require dividing the odd number \(v\) by a positive power of two.
After its initial \(5-d\) odd steps, that smaller parent reaches
\[
Y_d(t)=3^{5-d}v-1.
\]
Both \(X\) and \(Y_d\) are even. On a dyadic refinement
\(t=t_0+2^Mz\), suppose fixed source and parent words have lengths
\(h_s,h_p\) and odd-letter counts \(r_s,r_p\), respectively. Equality
of their affine coefficients in the free parameter \(z\) requires
\[
\frac{36\cdot2^M3^{5+r_s}}{2^{h_s}}
 =\frac{36\cdot2^M3^{5-d+r_p}}{2^{h_p}},
\qquad\text{so}\qquad
3^{d+r_s-r_p}=2^{h_s-h_p}.
\]
Unique factorization gives
\[
h_s=h_p=:h,
\qquad r_p-r_s=d.
\]
The odd-modulus restrictions leave infinitely many parameters on every
compatible dyadic refinement, by the Chinese remainder theorem. Thus an
affine equality there really does force equality of slopes.
If \(h=0\), then \(r_s=r_p=0\), already contradicting
\(r_p-r_s=d>0\). For the source, the first \(L+2\) tail letters are
\(01\,1^L\), so \(r_s=h-1\) whenever \(1\le h\le L+2\). The parent
tail begins even, so \(r_p\le h-1=r_s\), again a contradiction. This
proves the second assertion.

For the final claim, suppose that the displayed finite cover exists and let
\(H\) be the maximum of all direct-descent times and all source and
parent word lengths appearing in its certificates. Choose
\(L\ge\max\{6,H\}\). A direct-descent set cannot meet
\(\mathcal E_L\) by the first assertion. If a collision set meets
\(\mathcal E_L\), its dyadic congruence and the defining congruence of
\(\mathcal E_L\) are compatible; their intersection still contains
infinitely many parameters by the Chinese remainder theorem. The slope
argument then contradicts its fixed collision certificate. Thus every
\(C_i\) misses \(\mathcal E_L\), contrary to the fact that the \(C_i\)
cover \(C\supset\mathcal E_L\).
\end{proof}

The residues \(\tau_L\) are nested and converge in \(\mathbb Z_2\) to
\(-183/6561\). At that parameter, the time-ten affine state is \(-1\),
the 2-adic fixed point of the odd shortcut branch. The theorem therefore
identifies an exact method obstruction, not a divergent positive orbit:
its nested intersection need not contain a natural parameter. Parents
constructed after a different prefix, parameter-dependent joins not fixed
on a finite dyadic cover, or an unbounded-depth well-founded recursion
remain possible.

\section{Terminal powers}

% The valuation step is stated with its exact odd-length and positivity conditions.
\begin{lemma}[Divisibility at three]
For \(L>0\),
\[
3^{a+1}\mid2^L+1
\quad\Longleftrightarrow\quad
L\text{ is odd and }3^a\mid L.
\]
\end{lemma}
\begin{proof}
If \(3\mid2^L+1\), reduction modulo \(3\) makes \(L\) odd.
For odd \(L\), lifting the exponent gives
\(v_3(2^L+1)=1+v_3(L)\): apply the odd-prime sum-of-powers identity
with prime \(3\), bases \(2\) and \(1\), and odd exponent \(L\).
The prime divides \(2+1\) and does not divide either base. Comparing
the valuation to \(a+1\) proves both directions. The Lean companion
specializes this exact theorem from Mathlib \cite{Mathlib}.
\end{proof}

\begin{theorem}[Terminal-prefix classification]
Fix \(a\ge2\), let \(u>0\) be odd, and set \(n=2^a u-1\).
The legal prefix \(1^a01\) ends at a power of two if and only if
\[
u=\frac{2^{3^a t}+1}{3^{a+1}}
\quad\text{for some positive odd integer }t.
\]
Writing \(L=3^a t\), its endpoint is \(2^{L-2}\); the first hit of
\(1\) occurs at time \(a+L\). Its unique peak before that hit is
\((2^L-2)/3\), attained at time \(a\).
\end{theorem}
\begin{proof}
The odd-run formula gives \(T^a(n)=3^a u-1\), which is even.
If the following halving is odd, one more odd shortcut gives
\[
T^{a+2}(n)=\frac{3^{a+1}u-1}{4}.
\]
If this equals \(2^m\), then \(3^{a+1}u=2^{m+2}+1\).
The divisibility lemma forces \(m+2=3^a t\) with \(t>0\) odd and
recovers exactly the stated quotient for \(u\).

Conversely, the divisibility lemma makes that quotient integral. Its
numerator and denominator are odd, so \(u\) is odd, and it is positive.
Here \(L\ge9\). Reduction of \(3^{a+1}u=2^L+1\) modulo \(4\) gives
\(3^a u\equiv3\pmod4\); thus the halving after the first odd run is
odd, verifying the final letter. The endpoint is exactly \(2^{L-2}\).
It requires \(L-2\) further halvings to reach \(1\), with no earlier
hit on that tail. All states of the initial odd run increase strictly,
and its peak is \(3^a u-1=(2^L-2)/3\). The following half is smaller;
also \(3\cdot2^{L-2}<2^L-2\), so the first terminal power is smaller
than that peak. The subsequent powers decrease strictly. Every state
before the final \(1\) is greater than \(1\), proving both the unique
peak and the first-arrival assertion.
\end{proof}
